# ============================================================
#  GitHub Actions: build SalmanPDF untuk Windows, macOS, Linux
#  Berkas aktif: .github/workflows/build-release.yml
#  (berkas .txt ini salinan sumbernya untuk dibaca/diedit lokal)
#  Rilis: push tag v* (mis. v0.12.0) -> ketiga OS dibangun,
#  lalu SATU job "publish" mengumpulkan semua hasilnya dan
#  menerbitkannya sebagai GitHub Release berisi file unduhan.
#  Bisa juga dijalankan manual dari tab Actions (workflow_dispatch)
#  — run manual tidak menerbitkan rilis, hanya artefak.
#
#  Riwayat perbaikan: versi pertama gagal di langkah publikasi
#  (tiap job matrix berebut mengedit rilis yang sama + token tanpa
#  izin tulis eksplisit). Sekarang: permissions contents: write +
#  job publish tunggal setelah semua build selesai.
# ============================================================
name: build-salmanpdf

on:
  push:
    tags: ["v*"]
  workflow_dispatch:

permissions:
  contents: write

jobs:
  build:
    strategy:
      matrix:
        include:
          - os: windows-latest
            artifact: SalmanPDF-Windows
          - os: macos-latest
            artifact: SalmanPDF-macOS
          - os: ubuntu-latest
            artifact: SalmanPDF-Linux
    runs-on: ${{ matrix.os }}
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: "3.12"
      - name: Install dependencies
        run: |
          python -m pip install --upgrade pip
          pip install PySide6-Essentials PyMuPDF Pillow pyinstaller
      - name: Install pywin32 (Windows, backend MS Office)
        if: runner.os == 'Windows'
        run: pip install pywin32
      - name: Icons
        run: python make_icon.py
      - name: Build (Windows)
        if: runner.os == 'Windows'
        run: pyinstaller --noconfirm --windowed --name SalmanPDF --icon salmanpdf/assets/appicon.ico --exclude-module Xlib --exclude-module pymupdf.layout --exclude-module onnxruntime main.py
      - name: Build (macOS)
        if: runner.os == 'macOS'
        run: |
          mkdir -p build/SalmanPDF.iconset
          for s in 16 32 128 256 512; do sips -z $s $s salmanpdf/assets/appicon-1024.png --out build/SalmanPDF.iconset/icon_${s}x${s}.png >/dev/null; done
          for s in 32 64 256 512 1024; do h=$((s/2)); sips -z $s $s salmanpdf/assets/appicon-1024.png --out build/SalmanPDF.iconset/icon_${h}x${h}@2x.png >/dev/null; done
          iconutil -c icns build/SalmanPDF.iconset -o build/SalmanPDF.icns
          pyinstaller --noconfirm --windowed --name SalmanPDF --icon build/SalmanPDF.icns --exclude-module Xlib --exclude-module pymupdf.layout --exclude-module onnxruntime main.py
      - name: Build (Linux)
        if: runner.os == 'Linux'
        run: pyinstaller --noconfirm --windowed --name SalmanPDF --exclude-module pymupdf.layout --exclude-module onnxruntime main.py
      - name: Package (Windows)
        if: runner.os == 'Windows'
        run: Compress-Archive -Path dist/SalmanPDF -DestinationPath SalmanPDF-Windows.zip -Force
        shell: powershell
      - name: Package (macOS)
        if: runner.os == 'macOS'
        run: ditto -c -k --keepParent dist/SalmanPDF.app SalmanPDF-macOS.zip
      - name: Package (Linux)
        if: runner.os == 'Linux'
        run: cd dist && tar -czf ../SalmanPDF-Linux.tar.gz SalmanPDF
      - uses: actions/upload-artifact@v4
        with:
          name: ${{ matrix.artifact }}
          path: |
            SalmanPDF-Windows.zip
            SalmanPDF-macOS.zip
            SalmanPDF-Linux.tar.gz
          if-no-files-found: ignore

  publish:
    if: startsWith(github.ref, 'refs/tags/v')
    needs: build
    runs-on: ubuntu-latest
    steps:
      - name: Download all artifacts
        uses: actions/download-artifact@v4
        with:
          pattern: SalmanPDF-*
          merge-multiple: true
          path: release-files
      - name: Publish release
        uses: softprops/action-gh-release@v2
        with:
          files: |
            release-files/SalmanPDF-Windows.zip
            release-files/SalmanPDF-macOS.zip
            release-files/SalmanPDF-Linux.tar.gz
          fail_on_unmatched_files: true
